cmake_minimum_required(VERSION 3.24)
project(omni_tokenizer VERSION 0.1.0 LANGUAGES CXX)
option(OMNI_CUDA "Build the CUDA backend" OFF)
option(OMNI_BLAS "Use a system CBLAS library when available" ON)
option(OMNI_TESTS "Build native tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(omni SHARED src/model.cpp src/cpu.cpp src/runtime.cpp)
target_include_directories(omni PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(omni PRIVATE OMNI_BUILD)
set_target_properties(omni PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(omni PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/fp:precise;/utf-8>)
else()
  target_compile_options(omni PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
  target_link_libraries(omni PRIVATE OpenMP::OpenMP_CXX)
endif()
if(OMNI_BLAS)
  include(CheckIncludeFileCXX)
  find_package(BLAS QUIET)
  check_include_file_cxx(cblas.h OMNI_HAVE_CBLAS_H)
  if(BLAS_FOUND AND OMNI_HAVE_CBLAS_H)
    target_compile_definitions(omni PRIVATE OMNI_CBLAS)
    target_link_libraries(omni PRIVATE BLAS::BLAS)
  endif()
endif()
if(OMNI_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 80 86 89 90)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit 12.0 REQUIRED)
  target_sources(omni PRIVATE src/cuda.cu)
  target_compile_definitions(omni PRIVATE OMNI_CUDA_ENABLED)
  target_link_libraries(omni PRIVATE CUDA::cudart CUDA::cublas CUDA::cublasLt)
  set_target_properties(omni PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
endif()
include(GNUInstallDirs)
install(TARGETS omni EXPORT omniTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT omniTargets NAMESPACE omni:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/omni)
if(OMNI_TESTS)
  enable_testing()
  add_executable(omni_native_test tests/native_smoke.cpp)
  target_link_libraries(omni_native_test PRIVATE omni)
  add_test(NAME native_smoke COMMAND omni_native_test)
endif()
